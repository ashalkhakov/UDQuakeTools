object lamp_base : base {
	float brightness;
	void flicker( float times );
	boolean isLit() { return brightness > 0; }
};

void lamp_base::flicker( float times ) {
	while ( times > 0 ) {
		brightness = 1 - brightness;
		times--;
	}
}
